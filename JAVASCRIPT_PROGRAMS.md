### <a id="s1-reverse-a-string"></a>1. Reverse a String

```javascript
// Approach 1: Built-in methods
function reverseString(str) {
  return str.split('').reverse().join('');
}

// Approach 2: Loop (interviewers prefer this — shows logic)
function reverseString(str) {
  let reversed = '';
  for (let i = str.length - 1; i >= 0; i--) {
    reversed += str[i];
  }
  return reversed;
}

// Approach 3: Two pointers (in-place for arrays)
function reverseString(str) {
  const arr = str.split('');
  let left = 0, right = arr.length - 1;
  
  while (left < right) {
    [arr[left], arr[right]] = [arr[right], arr[left]]; // Swap
    left++;
    right--;
  }
  
  return arr.join('');
}

// Test:
console.log(reverseString('hello'));  // "olleh"
console.log(reverseString('JavaScript')); // "tpircSavaJ"

// Explanation:
// - split('') converts string to char array
// - reverse() reverses array in place
// - join('') converts back to string
// Time: O(n), Space: O(n)
```

### <a id="s2-check-if-a-string-is-palindrome"></a>2. Check if a String is Palindrome

```javascript
// A palindrome reads same forward and backward: "madam", "racecar"

// Approach 1: Compare with reverse
function isPalindrome(str) {
  const cleaned = str.toLowerCase().replace(/[^a-z0-9]/g, ''); // Remove non-alphanumeric
  return cleaned === cleaned.split('').reverse().join('');
}

// Approach 2: Two pointers (more efficient — no extra string created)
function isPalindrome(str) {
  const cleaned = str.toLowerCase().replace(/[^a-z0-9]/g, '');
  let left = 0, right = cleaned.length - 1;
  
  while (left < right) {
    if (cleaned[left] !== cleaned[right]) return false;
    left++;
    right--;
  }
  
  return true;
}

// Test:
console.log(isPalindrome('madam'));      // true
console.log(isPalindrome('A man a plan a canal Panama')); // true
console.log(isPalindrome('hello'));      // false

// Explanation:
// Two pointers start from both ends and move towards center
// If any mismatch found, it's NOT a palindrome
// Time: O(n), Space: O(1) for pointer approach
```

### <a id="s3-find-duplicate-characters-in-a-string"></a>3. Find Duplicate Characters in a String

```javascript
function findDuplicates(str) {
  const charCount = {};
  const duplicates = [];
  
  for (const char of str.toLowerCase()) {
    charCount[char] = (charCount[char] || 0) + 1;
  }
  
  for (const [char, count] of Object.entries(charCount)) {
    if (count > 1) {
      duplicates.push({ char, count });
    }
  }
  
  return duplicates;
}

// Test:
console.log(findDuplicates('programming'));
// [{char: 'r', count: 2}, {char: 'g', count: 2}, {char: 'm', count: 2}]

console.log(findDuplicates('hello world'));
// [{char: 'l', count: 3}, {char: 'o', count: 2}]

// Explanation:
// 1. Count frequency of each character using an object
// 2. Filter characters that appear more than once
// Time: O(n), Space: O(n)
```

### <a id="s4-find-first-non-repeating-character"></a>4. Find First Non-Repeating Character

```javascript
function firstNonRepeating(str) {
  const charCount = {};
  
  // Count all characters
  for (const char of str) {
    charCount[char] = (charCount[char] || 0) + 1;
  }
  
  // Find first with count 1
  for (const char of str) {
    if (charCount[char] === 1) return char;
  }
  
  return null; // All characters repeat
}

// Using Map (preserves insertion order)
function firstNonRepeating(str) {
  const map = new Map();
  
  for (const char of str) {
    map.set(char, (map.get(char) || 0) + 1);
  }
  
  for (const [char, count] of map) {
    if (count === 1) return char;
  }
  
  return null;
}

// Test:
console.log(firstNonRepeating('aabcbd'));  // 'c'
console.log(firstNonRepeating('aabbcc'));  // null
console.log(firstNonRepeating('javascript')); // 'j'

// Explanation:
// Two passes: first count all, then find first with count = 1
// We iterate string (not object) in second pass to maintain original order
// Time: O(n), Space: O(n)
```

### <a id="s5-check-two-strings-are-anagram-or-not"></a>5. Check Two Strings are Anagram or Not

```javascript
// Anagram: Same characters, different arrangement
// "listen" and "silent" → true

// Approach 1: Sort and compare
function isAnagram(str1, str2) {
  if (str1.length !== str2.length) return false;
  
  const sorted1 = str1.toLowerCase().split('').sort().join('');
  const sorted2 = str2.toLowerCase().split('').sort().join('');
  
  return sorted1 === sorted2;
}

// Approach 2: Character count (better — O(n) vs O(n log n))
function isAnagram(str1, str2) {
  if (str1.length !== str2.length) return false;
  
  const charCount = {};
  
  // Add counts for str1
  for (const char of str1.toLowerCase()) {
    charCount[char] = (charCount[char] || 0) + 1;
  }
  
  // Subtract counts for str2
  for (const char of str2.toLowerCase()) {
    if (!charCount[char]) return false; // Character not found or exhausted
    charCount[char]--;
  }
  
  return true;
}

// Test:
console.log(isAnagram('listen', 'silent')); // true
console.log(isAnagram('hello', 'world'));   // false
console.log(isAnagram('Astronomer', 'Moon starer')); // true (if spaces removed)

// Explanation:
// If two strings are anagrams, they have exactly same characters with same frequency
// Time: O(n), Space: O(n)
```

### <a id="s6-count-character-occurrence-in-a-string"></a>6. Count Character Occurrence in a String

```javascript
function countOccurrence(str, target) {
  let count = 0;
  for (const char of str.toLowerCase()) {
    if (char === target.toLowerCase()) count++;
  }
  return count;
}

// One-liner using split
function countOccurrence(str, target) {
  return str.toLowerCase().split(target.toLowerCase()).length - 1;
}

// Count ALL characters
function countAllChars(str) {
  return str.split('').reduce((acc, char) => {
    acc[char] = (acc[char] || 0) + 1;
    return acc;
  }, {});
}

// Test:
console.log(countOccurrence('hello world', 'l'));  // 3
console.log(countAllChars('javascript'));
// {j:1, a:2, v:1, s:1, c:1, r:1, i:1, p:1, t:1}

// Explanation:
// Simple iteration and counting
// split() approach: "hello".split('l') = ['he', '', 'o'] → length 3 - 1 = 2
// Time: O(n), Space: O(1) for single char, O(n) for all chars
```

### <a id="s7-reverse-words-in-a-sentence"></a>7. Reverse Words in a Sentence

```javascript
// "Hello World JavaScript" → "JavaScript World Hello"

function reverseWords(str) {
  return str.split(' ').reverse().join(' ');
}

// Handle multiple spaces
function reverseWords(str) {
  return str.trim().split(/\s+/).reverse().join(' ');
}

// Without built-in reverse (manual approach)
function reverseWords(str) {
  const words = str.trim().split(/\s+/);
  let left = 0, right = words.length - 1;
  
  while (left < right) {
    [words[left], words[right]] = [words[right], words[left]];
    left++;
    right--;
  }
  
  return words.join(' ');
}

// Test:
console.log(reverseWords('Hello World JavaScript')); // "JavaScript World Hello"
console.log(reverseWords('  the sky  is blue  '));   // "blue is sky the"

// Explanation:
// Split by space, reverse the array of words, join back
// /\s+/ handles multiple spaces between words
// Time: O(n), Space: O(n)
```

### <a id="s8-find-longest-word-in-a-string"></a>8. Find Longest Word in a String

```javascript
function longestWord(str) {
  const words = str.split(/\s+/);
  let longest = '';
  
  for (const word of words) {
    if (word.length > longest.length) {
      longest = word;
    }
  }
  
  return longest;
}

// Using reduce
function longestWord(str) {
  return str.split(/\s+/).reduce((longest, word) => 
    word.length > longest.length ? word : longest, ''
  );
}

// Test:
console.log(longestWord('The quick brown fox jumped')); // "jumped"
console.log(longestWord('I love JavaScript programming')); // "programming"

// Explanation:
// Split into words, track the longest one
// Time: O(n), Space: O(n) for split array
```

### <a id="s9-remove-duplicate-characters-from-string"></a>9. Remove Duplicate Characters from String

```javascript
// "programming" → "progamin"

// Approach 1: Set (preserves first occurrence order)
function removeDuplicates(str) {
  return [...new Set(str)].join('');
}

// Approach 2: Manual tracking
function removeDuplicates(str) {
  let result = '';
  const seen = {};
  
  for (const char of str) {
    if (!seen[char]) {
      seen[char] = true;
      result += char;
    }
  }
  
  return result;
}

// Test:
console.log(removeDuplicates('programming')); // "progamin"
console.log(removeDuplicates('aabbccdd'));    // "abcd"
console.log(removeDuplicates('hello'));       // "helo"

// Explanation:
// Set automatically removes duplicates and maintains insertion order
// Manual approach: track seen characters, only add first occurrence
// Time: O(n), Space: O(n)
```

```javascript
// Approach 1: Regex
function isOnlyDigits(str) {
  return /^\d+$/.test(str);

function isOnlyDigits(str) {
  if (str.length === 0) return false;
  
  for (const char of str) {
    if (char < '0' || char > '9') return false;
  }
  return true;
}

// Approach 3: Using isNaN
function isOnlyDigits(str) {
  return str.length > 0 && !isNaN(str) && !str.includes(' ');

// Explanation:
// ^\d+$ means: start(^) one or more digits(\d+) end($)
// Time: O(n), Space: O(1)
```

### <a id="s11-convert-string-to-title-case"></a>11. Convert String to Title Case

```javascript
// "hello world javascript" → "Hello World Javascript"

function toTitleCase(str) {
  return str.split(' ').map(word => 
    word.charAt(0).toUpperCase() + word.slice(1).toLowerCase()
  ).join(' ');

// Handle multiple spaces and edge cases
function toTitleCase(str) {
```

### <a id="s12-count-vowels-and-consonants"></a>12. Count Vowels and Consonants

```javascript
function countVowelsConsonants(str) {
  const vowels = 'aeiou';
  let vowelCount = 0, consonantCount = 0;
  
  for (const char of str.toLowerCase()) {
    if (char >= 'a' && char <= 'z') { // Only count alphabets
      if (vowels.includes(char)) vowelCount++;
      else consonantCount++;
    }
  }
  
  return { vowels: vowelCount, consonants: consonantCount };
}

// Test:
console.log(countVowelsConsonants('Hello World'));
// { vowels: 3, consonants: 7 }

console.log(countVowelsConsonants('JavaScript'));
// { vowels: 3, consonants: 7 }

// Explanation:
// Check if character is alphabet first, then check if vowel or consonant
// Skip spaces, numbers, special chars
// Time: O(n), Space: O(1)
```

### <a id="s13-find-substring-inside-a-string"></a>13. Find Substring Inside a String

```javascript
// Without using includes() or indexOf()

function findSubstring(str, sub) {
  if (sub.length > str.length) return -1;
  
  for (let i = 0; i <= str.length - sub.length; i++) {
    let found = true;
    
    for (let j = 0; j < sub.length; j++) {
      if (str[i + j] !== sub[j]) {
        found = false;
        break;
      }
    }
    
    if (found) return i; // Return starting index
  }
  
  return -1; // Not found
}

// Find all occurrences
function findAllOccurrences(str, sub) {
  const indices = [];
  let index = str.indexOf(sub);
  
  while (index !== -1) {
    indices.push(index);
    index = str.indexOf(sub, index + 1);
  }
  
  return indices;
}

// Test:
console.log(findSubstring('hello world', 'world'));  // 6
console.log(findSubstring('hello world', 'xyz'));    // -1
console.log(findAllOccurrences('abcabcabc', 'abc')); // [0, 3, 6]

// Explanation:
// Slide window of substring length across the main string
// Compare character by character at each position
// Time: O(n*m) where n = str length, m = sub length
```

### <a id="s14-check-string-rotation"></a>14. Check String Rotation

```javascript
// "abcde" rotated = "cdeab", "deabc", etc.
// Trick: If s2 is rotation of s1, then s2 exists in s1+s1!

function isRotation(str1, str2) {
  if (str1.length !== str2.length) return false;
  if (str1.length === 0) return true;
  
  const doubled = str1 + str1;
  return doubled.includes(str2);
}

// Test:

// "abcde", "bcdea", "cdeab", "deabc", "eabcd"
// Time: O(n), Space: O(n)
```

### <a id="s15-compress-string-aaabb--a3b2"></a>15. Compress String (aaabb → a3b2)

```javascript
function compressString(str) {
  let compressed = '';
  let count = 1;
  
  for (let i = 0; i < str.length; i++) {
    if (str[i] === str[i + 1]) {
      count++;
    } else {
      compressed += str[i] + count;
      count = 1;
    }
  }
  
  // Return compressed only if shorter than original
  return compressed.length < str.length ? compressed : str;
}

// Test:
console.log(compressString('aaabbbcc'));   // "a3b3c2"
console.log(compressString('aaabb'));      // "a3b2"
console.log(compressString('abc'));        // "abc" (compressed "a1b1c1" is longer!)

// Explanation:
// Walk through string, count consecutive same characters
// When current != next, append char + count and reset
// Time: O(n), Space: O(n)
```

### <a id="s17-find-all-permutations-of-string"></a>17. Find All Permutations of String

```javascript
// "abc" → ["abc", "acb", "bac", "bca", "cab", "cba"]

function getPermutations(str) {
  const result = [];
  
  if (str.length <= 1) return [str];
  
  function permute(current, remaining) {
    if (remaining.length === 0) {
      result.push(current);
      return;
    }
    
    for (let i = 0; i < remaining.length; i++) {
      const next = remaining[i];
      const newRemaining = remaining.slice(0, i) + remaining.slice(i + 1);
      permute(current + next, newRemaining);
    }
  }
  
  permute('', str);
  return result;
}

// Using swap approach
function permutations(str) {
  const result = [];
  const arr = str.split('');
  
  function swap(i, j) {
    [arr[i], arr[j]] = [arr[j], arr[i]];
  }
  
  function generate(n) {
    if (n === 1) {
      result.push(arr.join(''));
      return;
    }
    
    for (let i = 0; i < n; i++) {
      generate(n - 1);
      if (n % 2 === 0) swap(i, n - 1);
      else swap(0, n - 1);
    }
  }
  
  generate(arr.length);
  return result;
}

// Test:
console.log(getPermutations('abc'));
// ["abc", "acb", "bac", "bca", "cab", "cba"]

// Explanation:
// Recursive: Fix one character, find all permutations of remaining
```

### <a id="s20-check-balanced-brackets-in-string"></a>20. Check Balanced Brackets in String

```javascript
// "{[()]}" → true, "{[(])}" → false

function isBalanced(str) {
  const stack = [];
  const pairs = { ')': '(', ']': '[', '}': '{' };
  
  for (const char of str) {
    if (char === '(' || char === '[' || char === '{') {
      stack.push(char); // Opening bracket → push to stack
    } else if (char === ')' || char === ']' || char === '}') {
      if (stack.length === 0) return false; // Nothing to match with
      if (stack.pop() !== pairs[char]) return false; // Mismatch!
    }
  }
  
  return stack.length === 0; // Stack should be empty if all matched
}

// Test:

console.log(isBalanced('{[(])}')); // false
console.log(isBalanced('((()))'));  // true
console.log(isBalanced('({[}])'));  // false
console.log(isBalanced(''));        // true

// Explanation:
// Use stack: push opening brackets, pop when closing bracket found
// When popping, check if the popped bracket matches current closing bracket
// At end, stack must be empty (all brackets matched)
// Time: O(n), Space: O(n)
```

### <a id="a7-sort-array-without-sort"></a>7. Sort Array Without sort()

```javascript
// Bubble Sort (simplest to implement in interview)
function bubbleSort(arr) {
  const n = arr.length;
  
  for (let i = 0; i < n - 1; i++) {
    let swapped = false;
    
    for (let j = 0; j < n - i - 1; j++) {
      if (arr[j] > arr[j + 1]) {
        [arr[j], arr[j + 1]] = [arr[j + 1], arr[j]]; // Swap
        swapped = true;
      }
    }
    
    if (!swapped) break; // Array is already sorted
  }
  
  return arr;
}

// Selection Sort
function selectionSort(arr) {
  for (let i = 0; i < arr.length - 1; i++) {
    let minIdx = i;
    
    for (let j = i + 1; j < arr.length; j++) {
      if (arr[j] < arr[minIdx]) minIdx = j;
    }
    
    if (minIdx !== i) {
      [arr[i], arr[minIdx]] = [arr[minIdx], arr[i]];
    }
  }
  
  return arr;
}

// Quick Sort (best average performance)
function quickSort(arr) {
  if (arr.length <= 1) return arr;
  
  const pivot = arr[arr.length - 1];
  const left = [], right = [];
  
  for (let i = 0; i < arr.length - 1; i++) {
    if (arr[i] < pivot) left.push(arr[i]);
    else right.push(arr[i]);
  }
  
  return [...quickSort(left), pivot, ...quickSort(right)];
}

// Test:
console.log(bubbleSort([64, 34, 25, 12, 22, 11, 90]));
// [11, 12, 22, 25, 34, 64, 90]

// Explanation:
// Bubble Sort: Repeatedly swap adjacent elements if out of order
// Selection Sort: Find minimum, place at beginning, repeat
// Quick Sort: Pick pivot, partition into smaller/larger, recurse
// Time: Bubble/Selection O(n²), Quick Sort O(n log n) average
```

```javascript
function mergeSorted(arr1, arr2) {
  const merged = [];
  let i = 0, j = 0;
  
  while (i < arr1.length && j < arr2.length) {
    if (arr1[i] <= arr2[j]) {
      merged.push(arr1[i]);
      i++;
    } else {
      merged.push(arr2[j]);
      j++;
    }
  }
  
  // Add remaining elements
  while (i < arr1.length) merged.push(arr1[i++]);
  while (j < arr2.length) merged.push(arr2[j++]);
  
  return merged;
}

// When one array is exhausted, append remaining from other
```

### <a id="a14-find-largest-sum-of-subarray-kadanes-algorithm"></a>14. Find Largest Sum of Subarray (Kadane's Algorithm)

```javascript
// Find contiguous subarray with maximum sum

function maxSubarraySum(arr) {
  let currentMax = arr[0];
  let globalMax = arr[0];
  
  for (let i = 1; i < arr.length; i++) {
    // Either extend previous subarray or start new from current element
    currentMax = Math.max(arr[i], currentMax + arr[i]);
    globalMax = Math.max(globalMax, currentMax);
  }
  
  return globalMax;
}

// Also return the subarray
function maxSubarray(arr) {
  let currentMax = arr[0], globalMax = arr[0];
  let start = 0, end = 0, tempStart = 0;
  
  for (let i = 1; i < arr.length; i++) {
    if (arr[i] > currentMax + arr[i]) {
      currentMax = arr[i];
      tempStart = i; // New subarray starts here
    } else {
      currentMax = currentMax + arr[i];
    }
    
    if (currentMax > globalMax) {
      globalMax = currentMax;
      start = tempStart;
      end = i;
    }
  }
  
  return { sum: globalMax, subarray: arr.slice(start, end + 1) };
}

// Test:
console.log(maxSubarraySum([-2, 1, -3, 4, -1, 2, 1, -5, 4])); // 6
console.log(maxSubarray([-2, 1, -3, 4, -1, 2, 1, -5, 4]));
// { sum: 6, subarray: [4, -1, 2, 1] }

// Explanation:
// At each position, decide: extend current subarray OR start new one
// If current element alone is larger than current sum + element → start fresh
// Track the maximum sum seen so far
// Time: O(n), Space: O(1) — single pass!
```

```javascript
    } else {
      result.push(item);
    }
  }
  
  return result;

  );

function flatten(arr) {

      stack.push(...item); // Push nested items back to stack
    } else {
      result.unshift(item); // Add to front (maintains order)
    }
  }
  
  return result;
}
```

```javascript
    chunks[chunkIndex].push(item);
    return chunks;
  }, []);
}
```

### <a id="a20-find-longest-consecutive-sequence"></a>20. Find Longest Consecutive Sequence

```javascript
// [100, 4, 200, 1, 3, 2] → longest sequence: [1, 2, 3, 4] → length = 4

function longestConsecutive(arr) {
  if (arr.length === 0) return 0;
  
  const numSet = new Set(arr);
  let longestStreak = 0;
  
  for (const num of numSet) {
    // Only start counting from the beginning of a sequence
    if (!numSet.has(num - 1)) { // num is start of a sequence
      let currentNum = num;
      let currentStreak = 1;
      
      while (numSet.has(currentNum + 1)) {
        currentNum++;
        currentStreak++;
      }
      
      longestStreak = Math.max(longestStreak, currentStreak);
    }
  }
  
  return longestStreak;
}

// Test:
console.log(longestConsecutive([100, 4, 200, 1, 3, 2]));  // 4 (sequence: 1,2,3,4)
console.log(longestConsecutive([0, 3, 7, 2, 5, 8, 4, 6, 0, 1])); // 9 (0-8)

// Explanation:
// Key insight: Only start counting from the BEGINNING of a sequence
// A number is a sequence start if (num - 1) doesn't exist in Set
// Then count forward: num+1, num+2, ... until chain breaks
// Time: O(n), Space: O(n) — each number visited at most twice
```

### <a id="o4-deep-clone-an-object"></a>4. Deep Clone an Object

```javascript
// Shallow copy: Only top-level is copied, nested objects are still references
// Deep clone: Everything is copied, no shared references

// Approach 1: JSON (simple but has limitations)
function deepClone(obj) {
  return JSON.parse(JSON.stringify(obj));
}
// Limitations: Loses functions, undefined, Date becomes string, no circular refs

// Approach 2: structuredClone (modern, built-in)
function deepClone(obj) {
  return structuredClone(obj);
}
// Handles: Date, Map, Set, ArrayBuffer, circular refs
// Doesn't handle: Functions, DOM elements, Symbol

// Approach 3: Recursive (interview answer!)
function deepClone(obj) {
  if (obj === null || typeof obj !== 'object') return obj;
  
  if (obj instanceof Date) return new Date(obj);
  if (obj instanceof RegExp) return new RegExp(obj);
  if (Array.isArray(obj)) return obj.map(item => deepClone(item));
  
  const cloned = {};
  for (const key of Object.keys(obj)) {
    cloned[key] = deepClone(obj[key]); // Recursively clone each property
  }
  
  return cloned;
}

// Test:

  name: 'John', 
  address: { city: 'Mumbai', pin: 400001 },
  hobbies: ['reading', 'coding']
};

cloned.hobbies.push('gaming');

console.log(original.address.city); // 'Mumbai' (unchanged!)
console.log(original.hobbies);      // ['reading', 'coding'] (unchanged!)

// Explanation:
// Recursively visit every nested object/array and create NEW copies
// Primitives are copied by value automatically
// Objects/arrays need explicit recursion
```

### <a id="o6-compare-two-objects-for-equality"></a>6. Compare Two Objects for Equality

```javascript
// obj1 === obj2 won't work! (compares references, not values)

function deepEqual(obj1, obj2) {
  // Same reference
  if (obj1 === obj2) return true;
  
  // Null checks
  if (obj1 === null || obj2 === null) return false;
  if (typeof obj1 !== 'object' || typeof obj2 !== 'object') return false;
  
  // Compare keys
  const keys1 = Object.keys(obj1);
  const keys2 = Object.keys(obj2);
  
  if (keys1.length !== keys2.length) return false;
  
  // Compare each key's value recursively
  for (const key of keys1) {
    if (!keys2.includes(key)) return false;
    if (!deepEqual(obj1[key], obj2[key])) return false;
  }
  
  return true;
}

// Test:
console.log(deepEqual({ a: 1, b: { c: 2 } }, { a: 1, b: { c: 2 } })); // true
console.log(deepEqual({ a: 1, b: 2 }, { a: 1, b: 3 }));               // false
console.log(deepEqual([1, 2, 3], [1, 2, 3]));                          // true
console.log(deepEqual({ a: 1 }, { a: 1, b: 2 }));                     // false

// Explanation:
// Recursively compare every property at every level
// Two objects are equal if: same number of keys, same key names, same values (deep)
// Time: O(n) where n = total number of properties at all levels
```

### <a id="o9-merge-two-objects"></a>9. Merge Two Objects

```javascript
// Shallow merge
function mergeObjects(obj1, obj2) {
  return { ...obj1, ...obj2 }; // obj2 overwrites obj1 on conflicts
}

// Deep merge (nested objects merged, not overwritten)
function deepMerge(target, source) {
  const result = { ...target };
  
  for (const key of Object.keys(source)) {
    if (source[key] && typeof source[key] === 'object' && !Array.isArray(source[key])) {
      result[key] = deepMerge(result[key] || {}, source[key]);
    } else {
      result[key] = source[key];
    }
  }
  
  return result;
}

// Test:

// ↑ notifications REPLACED entirely (shallow merge!)

console.log(deepMerge(defaults, userPrefs));
// { theme: 'dark', fontSize: 14, notifications: { email: true, push: false } }
// ↑ notifications MERGED (deep merge!)

// Explanation:
// Shallow: spread operator, last value wins
// Deep: recursively merge nested objects
// Time: O(n) for shallow, O(n*depth) for deep
```

### <a id="o12-convert-nested-object-to-flat-object"></a>12. Convert Nested Object to Flat Object

```javascript
// { a: { b: { c: 1 } }, d: 2 } → { 'a.b.c': 1, d: 2 }

function flattenObject(obj, prefix = '', result = {}) {
  for (const [key, value] of Object.entries(obj)) {
    const newKey = prefix ? `${prefix}.${key}` : key;
    
    if (value && typeof value === 'object' && !Array.isArray(value)) {
      flattenObject(value, newKey, result); // Recurse into nested object
    } else {
      result[newKey] = value;
    }
  }
  
  return result;
}

// Unflatten (reverse operation)
function unflattenObject(obj) {
  const result = {};
  
  for (const [key, value] of Object.entries(obj)) {
    const keys = key.split('.');
    let current = result;
    
    for (let i = 0; i < keys.length - 1; i++) {
      if (!current[keys[i]]) current[keys[i]] = {};
      current = current[keys[i]];
    }
    
    current[keys[keys.length - 1]] = value;
  }
  
  return result;
}

// Test:

  user: {
    name: 'John',
    address: { city: 'Mumbai', pin: 400001 }
  },
  active: true
};

console.log(flattenObject(nested));
// { 'user.name': 'John', 'user.address.city': 'Mumbai', 'user.address.pin': 400001, active: true }

console.log(unflattenObject({ 'a.b.c': 1, 'd': 2 }));
// { a: { b: { c: 1 } }, d: 2 }

// Explanation:
// Recursively traverse nested object, build dot-notation keys
// When value is primitive (not object), add to result with full path key
// Time: O(n) where n = total number of leaf properties
```

```javascript
// Method 4: for...of with entries
for (const [key, value] of Object.entries(user)) {
  console.log(`${key}: ${value}`);
}

// Method 5: Object.getOwnPropertyNames() (includes non-enumerable)
Object.getOwnPropertyNames(user).forEach(key => {
  console.log(`${key}: ${user[key]}`);
});

// WHY avoid for...in?
// for...in iterates over INHERITED properties too (from prototype)
// Object.keys() only gives OWN enumerable properties (safer!)
```

### <a id="o19-transform-object-structure"></a>19. Transform Object Structure

```javascript
// Rename keys
function renameKeys(obj, keyMap) {
  return Object.fromEntries(
    Object.entries(obj).map(([key, value]) => [keyMap[key] || key, value])
  );
}

// Pick specific keys
function pick(obj, keys) {
  return keys.reduce((result, key) => {
    if (key in obj) result[key] = obj[key];
    return result;
  }, {});
}

// Omit specific keys
function omit(obj, keysToOmit) {
  return Object.fromEntries(
    Object.entries(obj).filter(([key]) => !keysToOmit.includes(key))
  );
}

// Map values
function mapValues(obj, fn) {
  return Object.fromEntries(
    Object.entries(obj).map(([key, value]) => [key, fn(value, key)])
  );
}
```

```javascript
function countProperties(obj) {
  let count = 0;
  
  for (const value of Object.values(obj)) {
    count++; // Count current property
    
    if (value && typeof value === 'object' && !Array.isArray(value)) {
      count += countProperties(value); // Recurse into nested objects
    }
  }
  
  return count;
}

function countLeafProperties(obj) {

  for (const value of Object.values(obj)) {

    } else {
      count++; // Only count non-object values (leaves)
    }
  }
  
  return count;
}
```

### <a id="m1-implement-debounce"></a>1. Implement Debounce

```javascript
// Debounce: Execute function AFTER user stops performing action for X ms
// Use case: Search input, window resize, button click (prevent double submit)

function debounce(fn, delay) {
  let timeoutId;
  
  return function(...args) {
    const context = this;
    
    clearTimeout(timeoutId); // Cancel previous scheduled execution
    
    timeoutId = setTimeout(() => {
      fn.apply(context, args); // Execute after delay
    }, delay);
  };
}

// With immediate option (execute on first call, then wait)
function debounce(fn, delay, immediate = false) {
  let timeoutId;
  
  return function(...args) {
    const context = this;
    const callNow = immediate && !timeoutId;
    
    clearTimeout(timeoutId);
    
    timeoutId = setTimeout(() => {
      timeoutId = null;
      if (!immediate) fn.apply(context, args);
    }, delay);
    
    if (callNow) fn.apply(context, args);
  };
}

// Test:
const expensiveSearch = debounce((query) => {
  console.log('Searching for:', query);
}, 300);

// User types: h-e-l-l-o (each keystroke calls expensiveSearch)
// Only ONE "Searching for: hello" is logged (after 300ms of no typing)

expensiveSearch('h');
expensiveSearch('he');
expensiveSearch('hel');
expensiveSearch('hell');
expensiveSearch('hello');
// After 300ms: "Searching for: hello" (only once!)

// Explanation:
// Every call resets the timer (clearTimeout)
// Function only executes when timer completes without interruption
// Time: O(1) per call, Space: O(1)
```

### <a id="m2-implement-throttle"></a>2. Implement Throttle

```javascript
// Throttle: Execute function at most ONCE per X ms
// Use case: Scroll handler, resize handler, shooting in game

function throttle(fn, interval) {
  let lastTime = 0;
  
  return function(...args) {
    const now = Date.now();
    
    if (now - lastTime >= interval) {
      lastTime = now;
      fn.apply(this, args);
    }
  };
}

// Throttle with trailing call (ensures last call is made)
function throttle(fn, interval) {
  let lastTime = 0;
  let timeoutId = null;
  
  return function(...args) {
    const now = Date.now();
    const remaining = interval - (now - lastTime);
    
    if (remaining <= 0) {
      if (timeoutId) {
        clearTimeout(timeoutId);
        timeoutId = null;
      }
      lastTime = now;
      fn.apply(this, args);
    } else if (!timeoutId) {
      timeoutId = setTimeout(() => {
        lastTime = Date.now();
        timeoutId = null;
        fn.apply(this, args);
      }, remaining);
    }
  };
}

// Test:
const throttledScroll = throttle(() => {
  console.log('Scroll position:', window.scrollY);
}, 200);

// Even if scroll fires 100 times per second,
// console.log only runs every 200ms (5 times per second)

// Difference from Debounce:
// Debounce: Wait for silence → execute once at the end
// Throttle: Execute regularly → at most once per interval

// Explanation:
// Track last execution time
// Only execute if enough time has passed since last execution
// Time: O(1) per call, Space: O(1)
```

### <a id="m3-polyfill-of-map--filter--reduce"></a>3. Polyfill of map / filter / reduce

```javascript
// ========= Array.prototype.map Polyfill =========
Array.prototype.myMap = function(callback, thisArg) {
  const result = [];
  
  for (let i = 0; i < this.length; i++) {
    if (i in this) { // Skip empty slots in sparse arrays
      result.push(callback.call(thisArg, this[i], i, this));
    }
  }
  
  return result;
};

// Test:
console.log([1, 2, 3].myMap(x => x * 2)); // [2, 4, 6]
console.log([1, 2, 3].myMap((x, i) => x + i)); // [1, 3, 5]

// ========= Array.prototype.filter Polyfill =========
Array.prototype.myFilter = function(callback, thisArg) {
  const result = [];
  
  for (let i = 0; i < this.length; i++) {
    if (i in this) {
      if (callback.call(thisArg, this[i], i, this)) {
        result.push(this[i]);
      }
    }
  }
  
  return result;
};

// Test:
console.log([1, 2, 3, 4, 5].myFilter(x => x > 3)); // [4, 5]
console.log([1, 2, 3, 4].myFilter(x => x % 2 === 0)); // [2, 4]

// ========= Array.prototype.reduce Polyfill =========
Array.prototype.myReduce = function(callback, initialValue) {
  let accumulator;
  let startIndex;
  
  if (initialValue !== undefined) {
    accumulator = initialValue;
    startIndex = 0;
  } else {
    if (this.length === 0) throw new TypeError('Reduce of empty array with no initial value');
    accumulator = this[0];
    startIndex = 1;
  }
  
  for (let i = startIndex; i < this.length; i++) {
    if (i in this) {
      accumulator = callback(accumulator, this[i], i, this);
    }
  }
  
  return accumulator;
};

// Test:
console.log([1, 2, 3, 4].myReduce((acc, cur) => acc + cur, 0)); // 10
console.log([1, 2, 3, 4].myReduce((acc, cur) => acc + cur));    // 10
console.log([[1,2],[3,4],[5]].myReduce((acc, cur) => acc.concat(cur), [])); // [1,2,3,4,5]

// Explanation:
// map: Transform each element, return new array (same length)
// filter: Keep elements that pass test, return new array (≤ same length)
// reduce: Accumulate all elements into single value (any type)
// All take callback(element, index, array) and optional thisArg
```

### <a id="m4-closure-based-counter"></a>4. Closure Based Counter

```javascript
// Basic counter
function createCounter() {
  let count = 0; // Private variable (closure)
  
  return {
    increment() { return ++count; },
    decrement() { return --count; },
    getCount() { return count; },
    reset() { count = 0; return count; }
  };
}

const counter = createCounter();
console.log(counter.increment()); // 1
console.log(counter.increment()); // 2
console.log(counter.decrement()); // 1
console.log(counter.getCount());  // 1
console.log(counter.reset());     // 0
// counter.count → undefined (can't access directly!)

// Counter with step
function createCounter(initialValue = 0, step = 1) {
  let count = initialValue;
  
  return {
    increment() { count += step; return count; },
    decrement() { count -= step; return count; },
    getValue() { return count; }
  };
}

const counter2 = createCounter(10, 5);
console.log(counter2.increment()); // 15
console.log(counter2.increment()); // 20
console.log(counter2.decrement()); // 15

// Function-based counter (returns next value on each call)
function makeCounter(start = 0) {
  let current = start;
  return () => current++;
}

const getId = makeCounter(1);
console.log(getId()); // 1
console.log(getId()); // 2
console.log(getId()); // 3

// Explanation:
// Closure keeps 'count' alive between function calls
// 'count' is private — accessible only through returned methods
// Each createCounter() call creates independent counter with own 'count'
```

### <a id="m8-call-apply-bind-implementation"></a>8. Call, Apply, Bind Implementation

```javascript
// call, apply, bind = methods to set 'this' context

const user = { name: 'John' };

function greet(greeting, punctuation) {
  return `${greeting}, ${this.name}${punctuation}`;
}

console.log(greet.call(user, 'Hello', '!'));    // "Hello, John!"

// ========= POLYFILLS (Interview favorites!) =========

// Function.prototype.myCall
Function.prototype.myCall = function(context, ...args) {
  context = context || globalThis; // Handle null/undefined
  const fnSymbol = Symbol(); // Unique key to avoid overwriting
  context[fnSymbol] = this;  // Attach function to context
  const result = context[fnSymbol](...args); // Call with context
  delete context[fnSymbol];  // Cleanup
  return result;
};

// Function.prototype.myApply
Function.prototype.myApply = function(context, args = []) {
  context = context || globalThis;
  const fnSymbol = Symbol();
  context[fnSymbol] = this;
  const result = context[fnSymbol](...args);
  delete context[fnSymbol];
  return result;
};

// Function.prototype.myBind
Function.prototype.myBind = function(context, ...boundArgs) {
  const fn = this; // Store original function
  
  return function(...callArgs) {
    return fn.apply(context, [...boundArgs, ...callArgs]);
  };
};

// Test polyfills:
console.log(greet.myCall(user, 'Hello', '!'));   // "Hello, John!"
console.log(greet.myApply(user, ['Hi', '?']));   // "Hi, John?"
const bound = greet.myBind(user, 'Hey');
console.log(bound('!!'));                         // "Hey, John!!"

// Explanation:
// call/apply: Temporarily attach function to context object, invoke, cleanup
// bind: Return new function that always calls original with fixed context
// Symbol used to avoid property name collisions
```

### <a id="m9-currying-function"></a>9. Currying Function

```javascript
// Currying: Transform f(a, b, c) into f(a)(b)(c)
// Each call returns a new function until all arguments received

// Basic currying
function curry(fn) {
  return function curried(...args) {
    if (args.length >= fn.length) {
      return fn.apply(this, args); // All args received, execute
    }
    return function(...args2) {
      return curried.apply(this, [...args, ...args2]); // Collect more args
    };
  };
}

// Example:
function add(a, b, c) {
  return a + b + c;
}

const curriedAdd = curry(add);
console.log(curriedAdd(1)(2)(3));   // 6
console.log(curriedAdd(1, 2)(3));   // 6
console.log(curriedAdd(1)(2, 3));   // 6
console.log(curriedAdd(1, 2, 3));   // 6

// Practical use cases:

// 1. Configuration functions
const log = curry((level, timestamp, message) => {
  console.log(`[${level}] ${timestamp}: ${message}`);
});

const errorLog = log('ERROR');       // Partially applied
const errorNow = errorLog(new Date().toISOString());
errorNow('Something failed');        // [ERROR] 2024-01-01T...: Something failed

// 2. Event handlers
const handleEvent = curry((eventType, element, handler) => {
  element.addEventListener(eventType, handler);
});

const onClick = handleEvent('click'); // Reusable for any element
onClick(button1, () => console.log('Button 1'));
onClick(button2, () => console.log('Button 2'));

// 3. Infinite currying (sum(1)(2)(3)...())
function sum(a) {
  return function(b) {
    if (b === undefined) return a; // No more args, return result
    return sum(a + b);
  };
}

console.log(sum(1)(2)(3)(4)()); // 10
// OR with valueOf:
function sum(a) {
  const fn = (b) => sum(a + b);
  fn.valueOf = () => a;
  return fn;
}
console.log(+sum(1)(2)(3)); // 6

// Explanation:
// Currying captures arguments one at a time via closures
// Each call returns a new function until enough args are collected
// Useful for: partial application, function composition, config
```

### <a id="m10-memoization-function"></a>10. Memoization Function

```javascript
// Memoization: Cache expensive function results based on arguments
// If same input → return cached output (no recomputation)

function memoize(fn) {
  const cache = new Map();
  
  return function(...args) {
    const key = JSON.stringify(args);
    
    if (cache.has(key)) {
      console.log('From cache');
      return cache.get(key);
    }
    
    console.log('Computing...');
    const result = fn.apply(this, args);
    cache.set(key, result);
    return result;
  };
}

// Test with expensive function:
function factorial(n) {
  if (n <= 1) return 1;
  return n * factorial(n - 1);
}

const memoizedFactorial = memoize(factorial);
console.log(memoizedFactorial(10)); // Computing... → 3628800
console.log(memoizedFactorial(10)); // From cache → 3628800 (instant!)
console.log(memoizedFactorial(5));  // Computing... → 120
console.log(memoizedFactorial(5));  // From cache → 120

// Fibonacci with memoization (transforms O(2^n) → O(n))
function memoizedFib() {
  const cache = {};
  
  function fib(n) {
    if (n in cache) return cache[n];
    if (n <= 1) return n;
    
    cache[n] = fib(n - 1) + fib(n - 2);
    return cache[n];
  }
  
  return fib;
}

const fib = memoizedFib();
console.log(fib(50)); // 12586269025 (instant! without memo: would take forever)

// Memoize with LRU (Least Recently Used) — bounded cache
function memoizeLRU(fn, maxSize = 100) {
  const cache = new Map();
  
  return function(...args) {
    const key = JSON.stringify(args);
    
    if (cache.has(key)) {
      const value = cache.get(key);
      // Move to end (most recently used)
      cache.delete(key);
      cache.set(key, value);
      return value;
    }
    
    const result = fn.apply(this, args);
    cache.set(key, result);
    
    // Evict oldest if over limit
    if (cache.size > maxSize) {
      const oldestKey = cache.keys().next().value;
      cache.delete(oldestKey);
    }
    
    return result;
  };
}

// Explanation:
// Cache stores: input args (as key) → computed result (as value)
// Before computing, check if result already cached
// Map preserves insertion order (useful for LRU)
// Time: O(1) for cached lookups, Space: O(n) for cache storage

// Real-world use: React.useMemo, reselect (Redux), API response caching
```

## Quick Reference: Key Patterns to Remember

| Pattern | When to Use | Time Complexity |
|---------|-------------|----------------|
| Two Pointers | Palindrome, reverse, sorted array pairs | O(n) |
| Hash Map/Set | Frequency count, duplicate detection, two sum | O(n) |
| Sliding Window | Substring problems, max subarray | O(n) |
| Stack | Balanced brackets, expression evaluation | O(n) |
| Recursion | Tree/nested structures, permutations | Varies |
| Sorting | When order matters, finding pairs | O(n log n) |
| Binary Search | Sorted array lookup | O(log n) |
| Reduce | Accumulate/transform collections | O(n) |
| Closure | Privacy, counters, memoization | O(1) |
| Curry/Compose | Function reuse, partial application | O(1) |
